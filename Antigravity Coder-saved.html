def process_approved_file(file_path):
    """Processes human-approved actions and triggers MCP calls"""
    metadata = parse_markdown_metadata(file_path)
    action_type = metadata.get("action")
    
    try:
        if action_type == "send_email":
            result = mock_send_email(metadata)
        elif action_type == "payment":
            result = mock_make_payment(metadata)
        
        # Move to completion folder
        move_to_done(file_path, success=True)
        log_action(action_type, "SUCCESS", metadata)
        
    except Exception as e:
        log_action(action_type, "FAILED", {"error": str(e)})
        move_to_rejected(file_path)

    for name, args in PROCESSES_TO_MONITOR.items():
        if not is_process_running(name):

            log_path = os.path.join(LOGS_PATH, f"{name}_stdout.log")
            
            with open(log_path, "a") as f:
                p = subprocess.Popen(
                    args, 
                    stdout=f, 
                    stderr=subprocess.STDOUT,
                    preexec_fn=os.setsid # Unix process isolation
                )
                save_pid(name, p.pid)

2

Configure MCP Servers

Link your external capabilities in mcp.json. Ensure paths to your email, browser, and banking MCP scripts are absolute and point to verified implementations.

3

Execute Watchdog

Run python scripts/watchdog.py. This command initiates the perpetual reasoning loop and starts monitoring for incoming perception events.
